cmake_minimum_required(VERSION 3.22)
project(VELUNO VERSION 0.1.0 LANGUAGES CXX)
option(VELUNO_STABLE_BUILD "Build an explicitly approved stable version" OFF)
if(VELUNO_STABLE_BUILD)
    set(VELUNO_APP_VERSION "${PROJECT_VERSION}")
    set(VELUNO_BUILD_CHANNEL "stable")
else()
    set(VELUNO_APP_VERSION "${PROJECT_VERSION}-dev")
    set(VELUNO_BUILD_CHANNEL "unstable")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
include(CTest)
include(GNUInstallDirs)
find_package(Qt6 6.4 REQUIRED COMPONENTS Core Gui Quick QuickControls2 Sql DBus Network)
find_package(PkgConfig REQUIRED)
find_package(OpenSSL REQUIRED COMPONENTS Crypto)
string(TIMESTAMP VELUNO_BUILD_UTC "%Y-%m-%dT%H:%M:%SZ" UTC)
pkg_check_modules(MPV REQUIRED IMPORTED_TARGET mpv)
qt_standard_project_setup()
add_library(veluno_platform STATIC src/platform/MediaLocation.cpp)
target_include_directories(veluno_platform PUBLIC src)
target_link_libraries(veluno_platform PUBLIC Qt6::Core Qt6::DBus)
add_library(veluno_database STATIC src/database/Database.cpp src/database/Migrations.cpp src/database/LibraryRepository.cpp)
target_include_directories(veluno_database PUBLIC src)
target_link_libraries(veluno_database PUBLIC Qt6::Core Qt6::Sql veluno_platform)
add_library(veluno_core STATIC src/core/AppSettings.cpp src/core/LanguageController.cpp)
target_include_directories(veluno_core PUBLIC src)
target_link_libraries(veluno_core PUBLIC veluno_database)
add_library(veluno_player STATIC src/player/MpvEngine.cpp src/player/VideoItem.cpp src/player/PlayerSettings.cpp)
target_include_directories(veluno_player PUBLIC src)
target_link_libraries(veluno_player PUBLIC Qt6::Quick veluno_database PRIVATE PkgConfig::MPV)
add_library(veluno_scanner STATIC src/scanner/Scanner.cpp src/scanner/FilenameParser.cpp)
target_include_directories(veluno_scanner PUBLIC src)
target_link_libraries(veluno_scanner PUBLIC veluno_database)
add_library(veluno_library STATIC src/library/LibraryController.cpp src/library/LibraryModels.cpp)
target_include_directories(veluno_library PUBLIC src)
target_link_libraries(veluno_library PUBLIC veluno_scanner Qt6::DBus)
add_library(veluno_personal STATIC src/profiles/PersonalController.cpp)
target_link_libraries(veluno_personal PUBLIC veluno_library veluno_player Qt6::Gui)
add_library(veluno_backup STATIC src/backup/BackupService.cpp)
target_link_libraries(veluno_backup PUBLIC veluno_database)
add_library(veluno_updater STATIC src/updater/UpdateService.cpp)
target_link_libraries(veluno_updater PUBLIC veluno_backup Qt6::Network OpenSSL::Crypto)
target_compile_definitions(veluno_updater PRIVATE
    VELUNO_BUILD_UTC="${VELUNO_BUILD_UTC}"
    VELUNO_APP_VERSION="${VELUNO_APP_VERSION}"
    VELUNO_BUILD_CHANNEL="${VELUNO_BUILD_CHANNEL}")
find_program(FFPROBE_EXECUTABLE ffprobe REQUIRED)
add_library(veluno_metadata STATIC src/metadata/MetadataController.cpp)
target_include_directories(veluno_metadata PUBLIC src)
target_link_libraries(veluno_metadata PUBLIC veluno_personal Qt6::Network)
qt_add_executable(veluno src/app/main.cpp)
target_link_libraries(veluno PRIVATE veluno_updater veluno_backup veluno_personal veluno_core veluno_player veluno_library veluno_scanner veluno_metadata Qt6::Gui Qt6::Quick Qt6::QuickControls2)
qt_add_resources(veluno ui PREFIX "/" FILES qml/Main.qml qml/themes/Theme.qml qml/components/NavButton.qml qml/pages/LibraryPage.qml qml/pages/SettingsPage.qml qml/player/PlayerPage.qml
    qml/components/PosterCard.qml qml/components/PosterShelf.qml qml/components/FolderPanel.qml
    qml/components/ScanPanel.qml qml/components/TitleDetails.qml qml/components/LibraryDialogs.qml qml/wizard/SetupWizard.qml
    qml/components/ContinueShelf.qml qml/pages/ProfilesPage.qml qml/pages/CollectionsPage.qml qml/pages/BackupPage.qml qml/pages/PlayerSettingsPage.qml qml/pages/UpdatesPage.qml)
install(PROGRAMS ${FFPROBE_EXECUTABLE} DESTINATION ${CMAKE_INSTALL_BINDIR})
install(TARGETS veluno RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
install(FILES packaging/veluno.desktop DESTINATION ${CMAKE_INSTALL_DATADIR}/applications)
install(FILES assets/veluno.svg DESTINATION ${CMAKE_INSTALL_DATADIR}/icons/hicolor/scalable/apps)
if(BUILD_TESTING)

    find_program(FFMPEG_EXECUTABLE ffmpeg REQUIRED)

    target_link_libraries(player_tests PRIVATE veluno_player Qt6::Test)

    set_tests_properties(player PROPERTIES TIMEOUT 60)

    if(WESTON_EXECUTABLE)

    endif()
    qt_add_executable(library_tests tests/LibraryTest.cpp)
    target_link_libraries(library_tests PRIVATE veluno_library Qt6::Test)

    target_link_libraries(network_tests PRIVATE veluno_library veluno_player Qt6::Test)

    qt_add_executable(database_tests tests/DatabaseTest.cpp)

    add_test(NAME qml COMMAND qml_tests -platform offscreen)
    set_tests_properties(qml PROPERTIES ENVIRONMENT "QT_QUICK_BACKEND=software;QML_DISABLE_DISK_CACHE=1")
