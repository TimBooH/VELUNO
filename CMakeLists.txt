cmake_minimum_required(VERSION 3.22)
project(VELUNO VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
include(CTest)
include(GNUInstallDirs)
find_package(Qt6 6.4 REQUIRED COMPONENTS Core Gui Quick QuickControls2 Sql)
find_package(PkgConfig REQUIRED)
pkg_check_modules(MPV REQUIRED IMPORTED_TARGET mpv)
qt_standard_project_setup()
add_library(veluno_database STATIC src/database/Database.cpp src/database/Migrations.cpp)
target_include_directories(veluno_database PUBLIC src)
target_link_libraries(veluno_database PUBLIC Qt6::Core Qt6::Sql)
add_library(veluno_core STATIC src/core/AppSettings.cpp)
target_include_directories(veluno_core PUBLIC src)
target_link_libraries(veluno_core PUBLIC veluno_database)
add_library(veluno_player STATIC src/player/MpvEngine.cpp)
target_include_directories(veluno_player PUBLIC src)
target_link_libraries(veluno_player PRIVATE PkgConfig::MPV)
add_library(veluno_scanner INTERFACE)
target_include_directories(veluno_scanner INTERFACE src)
target_link_libraries(veluno_scanner INTERFACE Qt6::Core)
add_library(veluno_metadata INTERFACE)
target_include_directories(veluno_metadata INTERFACE src)
target_link_libraries(veluno_metadata INTERFACE Qt6::Core)
qt_add_executable(veluno src/app/main.cpp)
target_link_libraries(veluno PRIVATE veluno_core veluno_player veluno_scanner veluno_metadata Qt6::Gui Qt6::Quick Qt6::QuickControls2)
qt_add_resources(veluno ui PREFIX "/" FILES qml/Main.qml qml/themes/Theme.qml qml/components/NavButton.qml qml/pages/LibraryPage.qml qml/pages/SettingsPage.qml)
install(TARGETS veluno RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
install(FILES packaging/veluno.desktop DESTINATION ${CMAKE_INSTALL_DATADIR}/applications)
install(FILES assets/veluno.svg DESTINATION ${CMAKE_INSTALL_DATADIR}/icons/hicolor/scalable/apps)
if(BUILD_TESTING)
    find_package(Qt6 REQUIRED COMPONENTS Test QuickTest)
    qt_add_executable(database_tests tests/DatabaseTest.cpp)
    target_link_libraries(database_tests PRIVATE veluno_core Qt6::Test)
    add_test(NAME database COMMAND database_tests)
    qt_add_executable(qml_tests tests/qml_main.cpp)
    target_compile_definitions(qml_tests PRIVATE QUICK_TEST_SOURCE_DIR="${CMAKE_CURRENT_SOURCE_DIR}/tests/qml")
    target_link_libraries(qml_tests PRIVATE veluno_core Qt6::QuickTest Qt6::QuickControls2)
    add_test(NAME qml COMMAND qml_tests -platform offscreen)
    set_tests_properties(qml PROPERTIES ENVIRONMENT "QT_QUICK_BACKEND=software;QML_DISABLE_DISK_CACHE=1")
endif()
